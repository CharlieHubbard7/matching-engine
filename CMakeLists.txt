cmake_minimum_required(VERSION 3.20)
project(matching_engine CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(ME_SANITIZE "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ME_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

add_library(me INTERFACE)
target_include_directories(me INTERFACE include)
target_compile_options(me INTERFACE -Wall -Wextra -Wpedantic)

include(CheckCXXCompilerFlag)
check_cxx_compiler_flag(-march=native HAS_MARCH_NATIVE)
if(HAS_MARCH_NATIVE)
  target_compile_options(me INTERFACE -march=native)
endif()

add_executable(gateway src/gateway.cpp)
add_executable(replay src/replay.cpp)
add_executable(bench bench/bench.cpp)
add_executable(itch_convert tools/itch_convert.cpp)
add_executable(test_engine tests/test_engine.cpp)
add_executable(test_gateway tests/test_gateway.cpp)
foreach(target gateway replay bench itch_convert test_engine test_gateway)
  target_link_libraries(${target} PRIVATE me)
endforeach()

enable_testing()
add_test(NAME engine COMMAND test_engine)
add_test(NAME gateway COMMAND test_gateway $<TARGET_FILE:gateway>)
